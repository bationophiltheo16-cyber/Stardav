  async function adminAddEvent(){
    const titre = document.getElementById('ev-titre').value.trim();
    const date = document.getElementById('ev-date').value;
    const lieu = document.getElementById('ev-lieu').value.trim();
    const prix = document.getElementById('ev-prix').value;
    const places = document.getElementById('ev-places').value;
    const photoFile = document.getElementById('ev-photo').files[0];
    const msg = document.getElementById('admin-add-msg');
    const btn = document.getElementById('ev-submit-btn');
    msg.innerHTML = '';

    if (!titre || !date || !lieu || !prix) {
      msg.innerHTML = `<div class="admin-msg err">Titre, date, lieu et prix sont obligatoires.</div>`;
      return;
    }

    btn.disabled = true; btn.textContent = 'Publication…';
    try {
      let imageUrl = null;

      if (photoFile) {
        const filename = `${Date.now()}-${photoFile.name.replace(/[^a-zA-Z0-9.]/g, '_')}`;
        const uploadRes = await fetch(
          `${SUPABASE_URL}/storage/v1/object/event-photos/${filename}`,
          {
            method: 'POST',
            headers: {
              Authorization: `Bearer ${adminToken}`,
              apikey: SUPABASE_ANON_KEY,
              'Content-Type': photoFile.type,
            },
            body: photoFile,
          }
        );
        if (!uploadRes.ok) {
          const errData = await uploadRes.json().catch(() => ({}));
          throw new Error(`Échec de l'envoi de la photo : ${errData.message || uploadRes.status}`);
        }
        imageUrl = `${SUPABASE_URL}/storage/v1/object/public/event-photos/${filename}`;
      }

      const insertRes = await fetch(`${SUPABASE_URL}/rest/v1/evenements`, {
        method: 'POST',
        headers: {
          Authorization: `Bearer ${adminToken}`,
          apikey: SUPABASE_ANON_KEY,
          'Content-Type': 'application/json',
          Prefer: 'return=representation',
        },
        body: JSON.stringify({
          titre,
          date_evenement: date,
          lieu,
          prix: Number(prix),
          places_disponibles: places ? Number(places) : null,
          image: imageUrl,
        }),
      });
      if (!insertRes.ok) {
        const errData = await insertRes.json().catch(() => ({}));
        throw new Error(errData.message || `Erreur ${insertRes.status}`);
      }

      msg.innerHTML = `<div class="admin-msg ok">✓ Événement publié avec succès.</div>`;
      document.getElementById('ev-titre').value = '';
      document.getElementById('ev-date').value = '';
      document.getElementById('ev-lieu').value = '';
      document.getElementById('ev-prix').value = '';
      document.getElementById('ev-places').value = '';
      document.getElementById('ev-photo').value = '';
      adminLoadEventsList();
      loadEvents();
    } catch (err) {
      msg.innerHTML = `<div class="admin-msg err">⚠️ ${err.message}</div>`;
    } finally {
      btn.disabled = false; btn.textContent = "Publier l'événement";
    }
  }

    if (!confirm('Supprimer définitivement cet événement ?')) return;
    try {
      const res = await fetch(`${SUPABASE_URL}/rest/v1/evenements?id=eq.${id}`, {
        method: 'DELETE',
        headers: { Authorization: `Bearer ${adminToken}`, apikey: SUPABASE_ANON_KEY },
      });
      if (!res.ok) throw new Error(`Erreur ${res.status}`);
      adminLoadEventsList();
      loadEvents();
    } catch (err) {
      alert("Erreur lors de la suppression : " + err.message);
    }
  }
